notepad test.js

const { Builder, By, until } = require("selenium-webdriver");

async function testBookApp() {

    const driver = await new Builder()
        .forBrowser("chrome")
        .build();

    try {

        // Open the application
        await driver.get("http://localhost:3000");

        // Wait until AngularJS loads the input
        const bookInput = await driver.wait(
            until.elementLocated(By.id("bookName")),
            10000
        );

        // Enter book name
        await bookInput.sendKeys("Java Programming");

        // Click Add Book button
        await driver.findElement(By.id("addBtn")).click();

        // Check the result
        const result = await driver.findElement(By.id("result"));

        await driver.wait(
            until.elementTextIs(
                result,
                "Book added: Java Programming"
            ),
            10000
        );

        console.log("TEST PASSED");
        console.log("Book added successfully!");

    } catch (error) {

        console.log("TEST FAILED:", error.message);

    } finally {

        await driver.quit();

    }
}

testBookApp();
