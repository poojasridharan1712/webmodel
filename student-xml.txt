transform.py


from lxml import etree

xml = etree.parse("student.xml")
xslt = etree.parse("student.xsl")

transform = etree.XSLT(xslt)
result = transform(xml)

with open("student.html", "wb") as file:
    file.write(
        etree.tostring(
            result,
            method="html",
            pretty_print=True,
            encoding="UTF-8"
        )
    )

print("XML transformed into HTML successfully!")
print("Open student.html in your browser.")
